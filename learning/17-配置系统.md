## 一、配置优先级

```
加载顺序（低 → 高）：
  ① ~/.mewcode/config.yaml        用户全局配置
  ② {project}/.mewcode/config.yaml  项目配置
  ③ {project}/.mewcode/config.local.yaml  本地覆盖（gitignore）

还可以用：
  - CLI 参数显式指定配置文件路径
  - MEWCODE_CONFIG 环境变量
```

### 合并策略

```java
private static void mergeConfig(AppConfig base, AppConfig override) {
    // providers → 完全替换（不合并）
    if (override.getProviders() != null) {
        base.setProviders(override.getProviders());
    }

    // mcp_servers → 按 name 合并（同名覆盖，新增追加）
    for (var s : override.getMcpServers()) {
        if (byName.containsKey(s.getName())) {
            servers.set(idx, s);  // 覆盖
        } else {
            servers.add(s);       // 新增
        }
    }

    // hooks → 追加（累加）
    base.getHooks().addAll(override.getHooks());

    // 沙箱配置 → 部分覆盖
    if (override.getSandbox() != null) {
        if (override.getSandbox().isEnabled()) base.getSandbox().setEnabled(true);
        if (override.getSandbox().isNetworkEnabled()) base.getSandbox().setNetworkEnabled(true);
    }
}
```

private static AppConfig loadSingleFile(Path configPath) {
    String content = Files.readString(configPath);
    var loaderOptions = new LoaderOptions();
    var constructor = new Constructor(AppConfig.class, loaderOptions);
    constructor.setPropertyUtils(new SnakeCasePropertyUtils());  // ← 蛇形命名支持
    var yaml = new Yaml(constructor);
    return yaml.load(content);

## 四、配置校验

```java
private static void validate(AppConfig cfg) throws ConfigException {
    for (var p : cfg.getProviders()) {
        // 必填字段检查
        if (isBlank(p.getName())) missing.add("name");
        if (isBlank(p.getProtocol())) missing.add("protocol");
        if (isBlank(p.getBaseUrl())) missing.add("base_url");
        if (isBlank(p.getModel())) missing.add("model");

        // 协议合法性检查
        if (!VALID_PROTOCOLS.contains(p.getProtocol())) {
            throw error("invalid protocol '%s', must be: anthropic, openai, openai-compat");
        }
    }
}
```

## 五、API Key 解析

```java
// ProviderConfig 中
public String resolvedApiKey() {
    if (apiKey != null && !apiKey.isBlank()) return apiKey;

    // 环境变量回退
    return switch (protocol) {
        case "anthropic"    → System.getenv("ANTHROPIC_API_KEY");
        case "openai"       → System.getenv("OPENAI_API_KEY");
        case "openai-compat" → System.getenv("OPENAI_API_KEY");
        default → "";
    };
}
```
